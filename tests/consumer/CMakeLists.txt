cmake_minimum_required(VERSION 3.21)
project(cell_consumer LANGUAGES CXX)
option(USE_CSS "Test optional CSS component" OFF)
if(USE_CSS)
    find_package(fltk-cell 0.1 CONFIG REQUIRED COMPONENTS css)
else()
    find_package(fltk-cell 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
if(USE_CSS)
    target_link_libraries(consumer PRIVATE fltk-cell::css)
else()
    target_link_libraries(consumer PRIVATE fltk-cell::fltk-cell)
endif()
enable_testing()
add_test(NAME installed-cell COMMAND consumer)
