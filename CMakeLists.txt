cmake_minimum_required(VERSION 3.21)
project(fltk-cell VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(FLTK_CELL_BUILD_EXAMPLE "Build the interactive gallery" ON)
option(FLTK_CELL_WITH_CSS "Build the optional fltk-css adapter" ON)
set(FLTK_CELL_CSS_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../fltk-css" CACHE PATH "Optional local fltk-css source")
include(FetchContent)
if(FLTK_CELL_WITH_CSS AND NOT TARGET fltk-css::fltk-css)
    find_package(fltk-css CONFIG QUIET)
    if(NOT TARGET fltk-css::fltk-css)
        set(FLTK_CSS_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
        if(EXISTS "${FLTK_CELL_CSS_SOURCE_DIR}/CMakeLists.txt")
            add_subdirectory("${FLTK_CELL_CSS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/css")
        else()
            FetchContent_Declare(cell_css GIT_REPOSITORY https://github.com/hdijkema/fltk-css.git
                GIT_TAG 5b936cca8ea009d5a82cfb90ae9ad6d0d4f66b78)
            FetchContent_MakeAvailable(cell_css)
        endif()
    endif()
endif()
if(NOT TARGET fltk::fltk)
    find_package(FLTK 1.4 CONFIG QUIET)
    if(NOT FLTK_FOUND)
        set(FLTK_BUILD_TEST OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_FLUID OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_FLTK_OPTIONS OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_GL OFF CACHE BOOL "" FORCE)
        set(FLTK_BUILD_FORMS OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(cell_fltk GIT_REPOSITORY https://github.com/fltk/fltk.git
            GIT_TAG 0d28470949323dedff75444d0011f0b5d38064f1)
        FetchContent_MakeAvailable(cell_fltk)
    endif()
endif()
add_library(fltk-cell STATIC src/fltk_cell.cpp src/fltk_cell_editor.cpp)
add_library(fltk-cell::fltk-cell ALIAS fltk-cell)
target_compile_features(fltk-cell PUBLIC cxx_std_17)
target_include_directories(fltk-cell PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(TARGET fltk::fltk-shared AND BUILD_SHARED_LIBS)
    target_link_libraries(fltk-cell PUBLIC fltk::fltk-shared fltk::images-shared)
else()
    target_link_libraries(fltk-cell PUBLIC fltk::fltk fltk::images)
endif()
if(MSVC)
    target_compile_options(fltk-cell PRIVATE /utf-8 /W4)
endif()
if(FLTK_CELL_WITH_CSS)
    add_library(fltk-cell-css INTERFACE)
    add_library(fltk-cell::css ALIAS fltk-cell-css)
    target_link_libraries(fltk-cell-css INTERFACE fltk-cell fltk-css::fltk-css)
endif()
if(FLTK_CELL_BUILD_EXAMPLE)
    add_executable(fltk-cell-gallery examples/gallery.cpp)
    target_link_libraries(fltk-cell-gallery PRIVATE fltk-cell)
    if(FLTK_CELL_WITH_CSS)
        target_link_libraries(fltk-cell-gallery PRIVATE fltk-cell-css)
        target_compile_definitions(fltk-cell-gallery PRIVATE FLTK_CELL_USE_CSS)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(fltk-cell-tests tests/contracts.cpp)
    target_link_libraries(fltk-cell-tests PRIVATE fltk-cell)
    add_test(NAME cell-contracts COMMAND fltk-cell-tests)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS fltk-cell EXPORT fltk-cell-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT fltk-cell-targets NAMESPACE fltk-cell:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-cell)
configure_package_config_file(cmake/fltk-cell-config.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/fltk-cell-config.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-cell)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/fltk-cell-config-version.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fltk-cell-config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/fltk-cell-config-version.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-cell)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fltk-cell)
